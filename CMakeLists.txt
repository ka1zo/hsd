cmake_minimum_required(VERSION 3.20)
project(HSD VERSION 0.1.0 LANGUAGES CXX)

find_package(sqlcipher CONFIG QUIET)
if(sqlcipher_FOUND)
    set(HSD_SQLCIPHER_TARGET sqlcipher::sqlcipher)
    get_target_property(HSD_SQLCIPHER_HINTS sqlcipher::sqlcipher INTERFACE_INCLUDE_DIRECTORIES)
    foreach(HSD_HINT IN LISTS HSD_SQLCIPHER_HINTS)
        if(EXISTS "${HSD_HINT}/sqlcipher/sqlite3.h")
            set(HSD_SQLCIPHER_HEADER_DIR "${HSD_HINT}/sqlcipher")
            break()
        elseif(EXISTS "${HSD_HINT}/sqlite3.h")
            set(HSD_SQLCIPHER_HEADER_DIR "${HSD_HINT}")
            break()
        endif()
    endforeach()
    if(NOT HSD_SQLCIPHER_HEADER_DIR)
        message(FATAL_ERROR "SQLCipher headers not found in the package include directories")
    endif()
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SQLCIPHER REQUIRED IMPORTED_TARGET sqlcipher)
    set(HSD_SQLCIPHER_TARGET PkgConfig::SQLCIPHER)
endif()

add_library(hsd_core STATIC cpp/src/models.cpp cpp/src/tokenizer.cpp cpp/src/lexical_index.cpp cpp/src/vector_index.cpp cpp/src/hybrid_index.cpp cpp/src/encrypted_store.cpp)
target_include_directories(hsd_core PUBLIC cpp/include)
if(sqlcipher_FOUND)
    target_include_directories(hsd_core PUBLIC ${HSD_SQLCIPHER_HEADER_DIR})
endif()
target_compile_features(hsd_core PUBLIC cxx_std_20)
target_compile_definitions(hsd_core PRIVATE SQLITE_HAS_CODEC)
target_link_libraries(hsd_core PUBLIC ${HSD_SQLCIPHER_TARGET})
set_target_properties(hsd_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

option(HSD_BUILD_PYTHON "Build the HSD Python extension" OFF)
if(HSD_BUILD_PYTHON)
    set(PYBIND11_FINDPYTHON ON)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(_core cpp/python/bindings.cpp)
    target_link_libraries(_core PRIVATE hsd_core)
    install(TARGETS _core LIBRARY DESTINATION hsd RUNTIME DESTINATION hsd)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(hsd_models_test tests/cpp/models_test.cpp)
    target_link_libraries(hsd_models_test PRIVATE hsd_core)
    add_test(NAME hsd_models COMMAND hsd_models_test)

    add_executable(hsd_tokenizer_test tests/cpp/tokenizer_test.cpp)
    target_link_libraries(hsd_tokenizer_test PRIVATE hsd_core)
    add_test(NAME hsd_tokenizer COMMAND hsd_tokenizer_test)

    add_executable(hsd_lexical_index_test tests/cpp/lexical_index_test.cpp)
    target_link_libraries(hsd_lexical_index_test PRIVATE hsd_core)
    add_test(NAME hsd_lexical_index COMMAND hsd_lexical_index_test)

    add_executable(hsd_vector_index_test tests/cpp/vector_index_test.cpp)
    target_link_libraries(hsd_vector_index_test PRIVATE hsd_core)
    add_test(NAME hsd_vector_index COMMAND hsd_vector_index_test)

    add_executable(hsd_hybrid_index_test tests/cpp/hybrid_index_test.cpp)
    target_link_libraries(hsd_hybrid_index_test PRIVATE hsd_core)
    add_test(NAME hsd_hybrid_index COMMAND hsd_hybrid_index_test)

    add_executable(hsd_encrypted_store_test tests/cpp/encrypted_store_test.cpp)
    target_compile_definitions(hsd_encrypted_store_test PRIVATE SQLITE_HAS_CODEC)
    target_link_libraries(hsd_encrypted_store_test PRIVATE hsd_core)
    add_test(NAME hsd_encrypted_store COMMAND hsd_encrypted_store_test)
endif()
